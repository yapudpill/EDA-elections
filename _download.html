paris_api_prefix <- "https://opendata.paris.fr/api/explore/v2.1/catalog/datasets/{dataset}/"
paris_api_export <- paste0(paris_api_prefix, "exports/{format}")
paris_api_attachement <- paste0(paris_api_prefix, "attachments/{att}")

dl_parquet <- function(dataset, filename, path) {
  if (!file_exists(path)) {
    message("Downloading ", filename)
    download.file(glue(paris_api_export, dataset = dataset, format = "parquet"), path, mode = "wb")
  }
}

dl_excel <- function(dataset, attachement, filename, path) {
  if (!file_exists(path)) {
    message("Downloading ", filename)
    download.file(glue(paris_api_attachement, dataset = dataset, att = attachement), path, mode = "wb")
  }
}
