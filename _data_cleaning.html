clean_pres <- function(df_brut){
  df_propre <- df_brut |>
    rename(any_of(c(
      tour = "numero_tour",
      nb_inscr = "nb_inscrit",
      nb_emarg = "nb_emargement",
      nb_exprim = "nb_exprime"
    )))
  
  #pour les votes blancs et nuls
  if ("nb_vote_blanc" %in% names(df_propre)) {
    df_propre <- df_propre |>
      mutate(nb_bl_nul = nb_vote_blanc + nb_vote_nul)
  }
  
  candidats_presents <- intersect(names(df_propre), partis_mapping$candidat)
  
  df_final <- df_propre |>
    select(
      id_bvote, annee, tour, nb_inscr, nb_votant, nb_bl_nul, nb_exprim, 
      all_of(candidats_presents)
    ) |>
    mutate(type_election = "presidentiel") |>
  
  pivot_longer(
      cols = all_of(candidats_presents),
      names_to = "candidat",
      values_to = "voix"
    ) |>
  left_join(partis_mapping, by = "candidat")
  return(df_final)
}
